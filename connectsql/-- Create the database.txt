-- Create the database
CREATE DATABASE bus_system;
USE bus_system;

-- Create the 'stops' table
CREATE TABLE stops (
    id INT AUTO_INCREMENT PRIMARY KEY,
    stopName VARCHAR(255) NOT NULL
);

-- Create the 'buses' table
CREATE TABLE buses (
    id INT AUTO_INCREMENT PRIMARY KEY,
    busNo VARCHAR(10) NOT NULL,
    total_stops INT NOT NULL,
    trips JSON NOT NULL
);

-- Create the 'routes' table
CREATE TABLE routes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    busNo VARCHAR(10) NOT NULL,
    stopName VARCHAR(255) NOT NULL,
    village VARCHAR(255),
    forword INT,
    reverse INT,
    latitude DECIMAL(10, 8),
    longitude DECIMAL(11, 8),
    FOREIGN KEY (busNo) REFERENCES buses(busNo)
);

CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(255) NOT NULL UNIQUE,
    email VARCHAR(255) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL
);


CREATE TABLE bus_reviews (
    bus_id INT,
    username VARCHAR(255),
    rating INT,
    review_text TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
